import random
import time
from collections import deque


def ölç(func, təkrar=5):
    """Funksiyanı bir neçə dəfə işlədib orta vaxtı (ms) qaytarır."""
    cəm = 0
    for _ in range(təkrar):
        başla = time.perf_counter()
        func()
        cəm += time.perf_counter() - başla
    return cəm / təkrar * 1000

# 1. ARRAY VƏ DYNAMIC ARRAY
# =====================================================
class StaticArray:
    """Sabit ölçülü array: ölçü əvvəlcədən verilir, böyümür."""

    def __init__(self, capacity):
        self.data = [None] * capacity
        self.size = 0

    def insert(self, index, value):
        if self.size == len(self.data):
            raise OverflowError("Array doludur!")
        for i in range(self.size, index, -1):   # elementləri sağa sürüşdür
            self.data[i] = self.data[i - 1]
        self.data[index] = value
        self.size += 1

    def delete(self, index):
        for i in range(index, self.size - 1):   # elementləri sola sürüşdür
            self.data[i] = self.data[i + 1]
        self.size -= 1

    def search(self, value):
        for i in range(self.size):
            if self.data[i] == value:
                return i
        return -1


def array_benchmark():
    print("\n--- Array (sabit) vs Dynamic Array (Python list), vaxt ms ilə ---")
    print(f"{'Həcm':>7} | {'Əməliyyat':<22} | {'Static':>9} | {'Dynamic':>9}")
    for n in (1_000, 10_000, 50_000):
        sa = StaticArray(n + 1)
        dyn = []
        for i in range(n):
            sa.insert(sa.size, i)
            dyn.append(i)

        # Sonuna əlavə: ikisi də O(1)
        t1 = ölç(lambda: (sa.insert(sa.size, 1), sa.delete(sa.size - 1)))
        t2 = ölç(lambda: (dyn.append(1), dyn.pop()))
        print(f"{n:>7} | {'Sona əlavə+silmə':<22} | {t1:>9.4f} | {t2:>9.4f}")

        # Əvvələ əlavə: ikisi də O(n)
        t1 = ölç(lambda: (sa.insert(0, 1), sa.delete(0)))
        t2 = ölç(lambda: (dyn.insert(0, 1), dyn.pop(0)))
        print(f"{n:>7} | {'Əvvələ əlavə+silmə':<22} | {t1:>9.4f} | {t2:>9.4f}")

        # Axtarış (ən pis hal: sonuncu element): O(n)
        t1 = ölç(lambda: sa.search(n - 1))
        t2 = ölç(lambda: dyn.index(n - 1))
        print(f"{n:>7} | {'Axtarış (sonuncu)':<22} | {t1:>9.4f} | {t2:>9.4f}")
        print("-" * 56)
    print("Qeyd: Python list C dilində yazıldığı üçün mütləq vaxt kiçikdir,")
    print("amma əsas məqam artımdır: sona əlavə sabit, əvvələ əlavə və axtarış həcmlə artır.")

# 2. SINGLY VƏ DOUBLY LINKED LIST
# =====================================================
class SNode:
    def __init__(self, data):
        self.data = data
        self.next = None


class SinglyLinkedList:
    def __init__(self):
        self.head = None          # tail yoxdur

    def prepend(self, data):      # O(1)
        node = SNode(data)
        node.next = self.head
        self.head = node

    def append(self, data):       # O(n): sona qədər getmək lazımdır
        node = SNode(data)
        if not self.head:
            self.head = node
            return
        cur = self.head
        while cur.next:
            cur = cur.next
        cur.next = node

    def search(self, data):       # O(n)
        cur = self.head
        while cur:
            if cur.data == data:
                return True
            cur = cur.next
        return False

    def delete(self, data):       # O(n)
        cur, prev = self.head, None
        while cur:
            if cur.data == data:
                if prev:
                    prev.next = cur.next
                else:
                    self.head = cur.next
                return True
            prev, cur = cur, cur.next
        return False


class DNode:
    def __init__(self, data):
        self.data = data
        self.next = None
        self.prev = None


class DoublyLinkedList:
    def __init__(self):
        self.head = None
        self.tail = None          # tail var

    def prepend(self, data):      # O(1)
        node = DNode(data)
        if not self.head:
            self.head = self.tail = node
        else:
            node.next = self.head
            self.head.prev = node
            self.head = node

    def append(self, data):       # O(1): tail sayəsində
        node = DNode(data)
        if not self.tail:
            self.head = self.tail = node
        else:
            node.prev = self.tail
            self.tail.next = node
            self.tail = node

    def search(self, data):       # O(n)
        cur = self.head
        while cur:
            if cur.data == data:
                return True
            cur = cur.next
        return False

    def delete_last(self):        # O(1): tail.prev var
        if not self.tail:
            return
        if self.tail.prev:
            self.tail = self.tail.prev
            self.tail.next = None
        else:
            self.head = self.tail = None

    def delete(self, data):       # O(n) axtarış, sonra O(1) silmə
        cur = self.head
        while cur:
            if cur.data == data:
                if cur.prev:
                    cur.prev.next = cur.next
                else:
                    self.head = cur.next
                if cur.next:
                    cur.next.prev = cur.prev
                else:
                    self.tail = cur.prev
                return True
            cur = cur.next
        return False


def linked_list_benchmark():
    n = 3000
    print(f"\n--- Singly vs Doubly Linked List ({n} element), vaxt ms ilə ---")
    print(f"{'Əməliyyat':<28} | {'Singly':>9} | {'Doubly':>9}")

    def doldur(cls):
        lst = cls()
        for i in range(n):
            lst.prepend(i)
        return lst

    s, d = doldur(SinglyLinkedList), doldur(DoublyLinkedList)

    t1 = ölç(lambda: (s.prepend(-1), s.delete(-1)))
    t2 = ölç(lambda: (d.prepend(-1), d.delete(-1)))
    print(f"{'Əvvələ əlavə (prepend)':<28} | {t1:>9.4f} | {t2:>9.4f}")

    t1 = ölç(lambda: (s.append(-1), s.delete(-1)))
    t2 = ölç(lambda: (d.append(-1), d.delete_last()))
    print(f"{'Sona əlavə (append)+silmə':<28} | {t1:>9.4f} | {t2:>9.4f}")

    t1 = ölç(lambda: s.search(n))      # n yoxdur -> bütün siyahı gəzilir
    t2 = ölç(lambda: d.search(n))
    print(f"{'Axtarış (ən pis hal)':<28} | {t1:>9.4f} | {t2:>9.4f}")
    print("Nəticə: Doubly list-də tail olduğu üçün sona əlavə O(1)-dir,")
    print("Singly-də isə sona qədər getmək lazımdır, O(n). Əvəzində Singly daha az yaddaş işlədir.")

# 3. HASH TABLE (CHAINING)
# =====================================================
class HashTable:
    def __init__(self, capacity=7):
        self.capacity = capacity
        self.buckets = [[] for _ in range(capacity)]

    def _index(self, key):
        return hash(key) % self.capacity

    def insert(self, key, value):
        bucket = self.buckets[self._index(key)]
        for pair in bucket:
            if pair[0] == key:          # eyni açar varsa yenilə
                pair[1] = value
                return
        bucket.append([key, value])     # collision olsa, zəncirə əlavə olunur

    def search(self, key):
        for k, v in self.buckets[self._index(key)]:
            if k == key:
                return v
        return None

    def delete(self, key):
        bucket = self.buckets[self._index(key)]
        for i, pair in enumerate(bucket):
            if pair[0] == key:
                del bucket[i]
                return True
        return False

    def items(self):
        return [(k, v) for bucket in self.buckets for k, v in bucket]

    def show(self):
        for i, bucket in enumerate(self.buckets):
            zəncir = " -> ".join(f"[{k}: {v}]" for k, v in bucket) or "boş"
            print(f"  Bucket {i}: {zəncir}")


def hash_demo():
    print("\n--- Hash Table (chaining) nümayişi ---")
    ht = HashTable(capacity=5)
    for i in range(1, 9):               # 8 element, 5 bucket -> collision qaçılmazdır
        ht.insert(i, f"Tapşırıq {i}")
    ht.show()
    print("Axtarış (3):", ht.search(3))
    ht.delete(3)
    print("Silindikdən sonra axtarış (3):", ht.search(3))

# 4. BINARY SEARCH TREE (iterativ, rekursiya limiti problemi yoxdur)
# =====================================================
class BSTNode:
    def __init__(self, key, value=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None


class BST:
    def __init__(self):
        self.root = None

    def insert(self, key, value=None):
        node = BSTNode(key, value)
        if not self.root:
            self.root = node
            return
        cur = self.root
        while True:
            if key == cur.key:
                cur.value = value
                return
            if key < cur.key:
                if not cur.left:
                    cur.left = node
                    return
                cur = cur.left
            else:
                if not cur.right:
                    cur.right = node
                    return
                cur = cur.right

    def search(self, key):
        cur = self.root
        while cur and cur.key != key:
            cur = cur.left if key < cur.key else cur.right
        return cur

    def delete(self, key):
        parent, cur = None, self.root
        while cur and cur.key != key:
            parent = cur
            cur = cur.left if key < cur.key else cur.right
        if not cur:
            return False
        if cur.left and cur.right:      # 3-cü hal: iki uşaq -> inorder successor
            sp, s = cur, cur.right
            while s.left:
                sp, s = s, s.left
            cur.key, cur.value = s.key, s.value
            parent, cur = sp, s         # indi successor silinəcək (ən çox 1 uşağı var)
        child = cur.left or cur.right   # 1-ci və 2-ci hal: yarpaq və ya bir uşaq
        if parent is None:
            self.root = child
        elif parent.left is cur:
            parent.left = child
        else:
            parent.right = child
        return True

    def inorder(self):
        """Açarları artan sırada qaytarır."""
        nəticə, stack, cur = [], [], self.root
        while stack or cur:
            while cur:
                stack.append(cur)
                cur = cur.left
            cur = stack.pop()
            nəticə.append((cur.key, cur.value))
            cur = cur.right
        return nəticə

    def height(self):
        if not self.root:
            return 0
        yüksəklik, stack = 0, [(self.root, 1)]
        while stack:
            node, d = stack.pop()
            yüksəklik = max(yüksəklik, d)
            if node.left:
                stack.append((node.left, d + 1))
            if node.right:
                stack.append((node.right, d + 1))
        return yüksəklik


def bst_demo():
    n = 1000
    print(f"\n--- BST: sıralı vs təsadüfi verilənlər ({n} element) ---")
    sıralı = list(range(n))
    təsadüfi = sıralı[:]
    random.shuffle(təsadüfi)

    nəticələr = {}
    for ad, data in (("Sıralı", sıralı), ("Təsadüfi", təsadüfi)):
        t = BST()
        for x in data:
            t.insert(x)
        vaxt = ölç(lambda: [t.search(x) for x in range(n)], təkrar=3)
        nəticələr[ad] = (t.height(), vaxt)

    print(f"{'Verilən':<10} | {'Hündürlük':>10} | {n} axtarışın vaxtı (ms)")
    for ad, (h, v) in nəticələr.items():
        print(f"{ad:<10} | {h:>10} | {v:>10.3f}")
    print("Nəticə: sıralı əlavədə hər element əvvəlkinin sağına düşür,")
    print("ağac linked list-ə çevrilir (hündürlük n, axtarış O(n)).")
    print("Təsadüfi verilənlərdə ağac balanslı qalır (hündürlük təxminən log n, axtarış O(log n)).")

    print("\nSilmə nümunəsi (50, 30, 70, 20, 40, 60, 80):")
    t = BST()
    for x in (50, 30, 70, 20, 40, 60, 80):
        t.insert(x)
    for x, hal in ((20, "yarpaq"), (30, "bir uşaq"), (50, "iki uşaq")):
        t.delete(x)
        print(f"  {x} silindi ({hal}) -> {[k for k, _ in t.inorder()]}")

# 5. TASK MANAGER (Stack, Queue, Deque, Hash Table, BST real istifadədə)
# =====================================================
class TaskManager:
    def __init__(self):
        self.tasks = HashTable(capacity=11)  # id -> başlıq (sürətli axtarış)
        self.sorted_ids = BST()              # id-lər sıralı saxlanılır
        self.queue = deque()                 # Queue (FIFO): icra növbəsi
        self.undo_stack = []                 # Stack (LIFO): Undo
        self.history = deque(maxlen=5)       # Deque: son 5 əməliyyat
        self.next_id = 1

    def _add(self, task_id, title):
        self.tasks.insert(task_id, title)
        self.sorted_ids.insert(task_id, title)
        self.queue.append(task_id)

    def _remove(self, task_id):
        self.tasks.delete(task_id)
        self.sorted_ids.delete(task_id)
        if task_id in self.queue:
            self.queue.remove(task_id)

    def add(self, title):
        task_id = self.next_id
        self.next_id += 1
        self._add(task_id, title)
        self.undo_stack.append(("add", task_id, title))
        self.history.append(f"Əlavə edildi: #{task_id} {title}")
        return task_id

    def delete(self, task_id):
        title = self.tasks.search(task_id)
        if title is None:
            return False
        self._remove(task_id)
        self.undo_stack.append(("delete", task_id, title))
        self.history.append(f"Silindi: #{task_id} {title}")
        return True

    def process_next(self):
        if not self.queue:
            return None
        task_id = self.queue.popleft()      # ən əvvəl gələn ilk icra olunur
        title = self.tasks.search(task_id)
        self.tasks.delete(task_id)
        self.sorted_ids.delete(task_id)
        self.history.append(f"İcra edildi: #{task_id} {title}")
        return task_id, title

    def undo(self):
        if not self.undo_stack:
            return None
        action, task_id, title = self.undo_stack.pop()   # ən son əməliyyat
        if action == "add":
            self._remove(task_id)
        else:
            self._add(task_id, title)
        self.history.append(f"Undo: {action} #{task_id}")
        return action, task_id, title

def oxu_id(mesaj):

        return int(input(mesaj))
    except ValueError:

        return None


def main():

    while True:
        print(MENYU)
        seçim = input("Seçim: ").strip()

        if seçim == "1":
            başlıq = input("Tapşırıq başlığı: ").strip()

        elif seçim == "7":
